# Apply as a Cloudflare Single Redirect before switching the production origin.
# This preserves the legacy WordPress product archive URL seen in search results.

Rule name: Legacy product archive to governed catalogue
Expression: (http.request.uri.path eq "/products/20325959") or (http.request.uri.path eq "/products/20325959/") or (http.request.uri.path eq "/products/")
Target URL: concat("https://jftagro.com/products.html", if(len(http.request.uri.query) > 0, concat("?", http.request.uri.query), ""))
Status code: 301
Preserve query string: Yes (handled by target expression)
